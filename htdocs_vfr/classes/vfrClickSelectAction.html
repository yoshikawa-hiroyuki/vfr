Description

vfrClickSelectActionクラスは、描画領域(vfrDrawArea)への マウスクリックによるオブジェクト選択を行う デフォルトアクションのクラスである。

vfrClickSelectActionクラスのインスタンスはプログラム内で1つだけ存在し、 その取得にはinstance()メソッドを用いる。

Behavior

vfrClickSelectActionクラスは、execute()メソッドがコールされると、 以下の動作を行う。
  • パスされた仮想イベントから、イベント発生時のマウス位置を取得する
  • 既に選択されていたオブジェクトが存在する場合は、バウンディングボックスを非表示にし、 選択を解除する
  • マウス位置について、クリックセレクションテストを実行する
  • セレクションテストに合格したオブジェクトがある場合は、その中で視点に最も近いオブジェクトを 選択されたオブジェクトとして登録し、バウンディングボックスを表示させ、 vfrEvSDrag, vfrEvCDrag, vfrEvSCDragのアクションをそれぞれ vfrTransObjAction, vfrRotObjAction, vfrScaleObjAction に設定する (仮想イベント割り当てボタンタイプにより、設定するイベントタイプは変更される)
  • セレクションテストに合格したオブジェクトがない場合は、 vfrEvSDrag, vfrEvCDrag, vfrEvSCDragのアクションをそれぞれ vfrTransCameraAction, vfrRotCameraAction, vfrScaleCameraAction に設定する (仮想イベント割り当てボタンタイプにより、設定するイベントタイプは変更される)
これらの動作は、以下のコードで実装されている。 
void
vfrClickSelectAction::execute(vfrEvent& e) {
    if ( ! e.isClickEvent() ) return;

    Point2   cp;
    vfrScreen* screen = (vfrScreen*)e.getScreen();
    if ( ! screen ) return;
    if ( ! e.getMPoint(&cp) ) return;
    cp = screen->getRelativePoint(cp);
    unsigned int *r = screen->clickSelect(cp.x, cp.y);
    vfrObject* selectedObj = vfrDefaultActions::GetSelectedObj();
    if ( selectedObj ) {
        selectedObj->setBboxShowMode(FALSE);
        vfrDefaultActions::SetSelectedObj(NULL);
    }
    if ( r[0] > 0 ) {
        vfrObject* co = screen->getObj(r[1]);
        if ( co ) {
            screen->clickFeedback(cp.x, cp.y, r[1]);
            co->setBboxShowMode(TRUE);
            vfrDefaultActions::SetSelectedObj(co);

            // Change Actions
            vfrDrawArea* da = screen->getParent();
            if ( ! da ) return;
            vfrDispatch& dpr = vfrDispatch::instance(*da);
            if ( _bType & VFRDA_MBLEFT ) {
                vfrEvent& Sdrag = vfrEvSDrag::instance(dpr);
                Sdrag.regist(&vfrTransObjAction::instance());
                vfrEvent& Cdrag = vfrEvCDrag::instance(dpr);
                Cdrag.regist(&vfrRotObjAction::instance());
                vfrEvent& SCdrag = vfrEvSCDrag::instance(dpr);
                SCdrag.regist(&vfrScaleObjAction::instance());
            }
            if ( _bType & VFRDA_MBRIGHT ) {
                vfrEvent& Sdrag = vfrEvRSDrag::instance(dpr);
                Sdrag.regist(&vfrTransObjAction::instance());
                vfrEvent& Cdrag = vfrEvRCDrag::instance(dpr);
                Cdrag.regist(&vfrRotObjAction::instance());
                vfrEvent& SCdrag = vfrEvRSCDrag::instance(dpr);
              SCdrag.regist(&vfrScaleObjAction::instance());
            }
            if ( _bType & VFRDA_MBMIDDLE ) {
                vfrEvent& Sdrag = vfrEvMSDrag::instance(dpr);
                Sdrag.regist(&vfrTransObjAction::instance());
                vfrEvent& Cdrag = vfrEvMCDrag::instance(dpr);
                Cdrag.regist(&vfrRotObjAction::instance());
                vfrEvent& SCdrag = vfrEvMSCDrag::instance(dpr);
                SCdrag.regist(&vfrScaleObjAction::instance());
            }

            return;
        }
    }
    vfrDrawArea* da = screen->getParent();
    if ( ! da ) return;
    vfrDispatch& dpr = vfrDispatch::instance(*da);
    if ( _bType & VFRDA_MBLEFT ) {
        vfrEvent& Sdrag = vfrEvSDrag::instance(dpr);
        Sdrag.regist(&vfrTransCameraAction::instance());
        vfrEvent& Cdrag = vfrEvCDrag::instance(dpr);
        Cdrag.regist(&vfrRotCameraAction::instance());
        vfrEvent& SCdrag = vfrEvSCDrag::instance(dpr);
        SCdrag.regist(&vfrScaleCameraAction::instance());
    }
    if ( _bType & VFRDA_MBRIGHT ) {
        vfrEvent& Sdrag = vfrEvRSDrag::instance(dpr);
        Sdrag.regist(&vfrTransCameraAction::instance());
        vfrEvent& Cdrag = vfrEvRCDrag::instance(dpr);
        Cdrag.regist(&vfrRotCameraAction::instance());
        vfrEvent& SCdrag = vfrEvRSCDrag::instance(dpr);
        SCdrag.regist(&vfrScaleCameraAction::instance());
    }
    if ( _bType & VFRDA_MBMIDDLE ) {
        vfrEvent& Sdrag = vfrEvMSDrag::instance(dpr);
        Sdrag.regist(&vfrTransCameraAction::instance());
        vfrEvent& Cdrag = vfrEvMCDrag::instance(dpr);
        Cdrag.regist(&vfrRotCameraAction::instance());
        vfrEvent& SCdrag = vfrEvMSCDrag::instance(dpr);
        SCdrag.regist(&vfrScaleCameraAction::instance());
    }
}
